cmake_minimum_required(VERSION 3.19)
project(AeroCore VERSION 0.7.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CTest)
option(AEROCORE_CORE_ONLY "Build portable protocol/math tests without Qt" OFF)
add_library(AeroCore_Telemetry STATIC MavlinkReadOnly.cpp MavlinkReadOnly.hpp SimulationProtocol.hpp)
target_include_directories(AeroCore_Telemetry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(BUILD_TESTING)
    foreach(test mavlink_readonly_test simulation_protocol_test experiment_core_test)
        add_executable(${test} tests/${test}.cpp)
        target_link_libraries(${test} PRIVATE AeroCore_Telemetry)
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_FOUND)
        foreach(test launcher_test profile_test fleet_test)
            add_test(NAME ${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${test}.py)
        endforeach()
    endif()
endif()
if(AEROCORE_CORE_ONLY)
    return()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
add_library(AeroCore_Core STATIC LabTypes.hpp LabTypes.cpp ExperimentCore.hpp)
target_link_libraries(AeroCore_Core PUBLIC Qt6::Core)
target_include_directories(AeroCore_Core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(AeroCore_Simulation STATIC SimulatorSession.hpp SimulatorSession.cpp LabTransport.hpp LabTransport.cpp ExperimentRunner.hpp ExperimentRunner.cpp FlightState.hpp)
target_link_libraries(AeroCore_Simulation PUBLIC AeroCore_Core AeroCore_Telemetry Qt6::Network)
add_library(AeroCore_UI STATIC sitlpanel.h sitlpanel.cpp sitlpanel.ui LabPanel.hpp LabPanel.cpp labpanel.ui LabChart.hpp LabChart.cpp FlightViews.hpp FlightViews.cpp FleetPanel.hpp FleetPanel.cpp SessionRecording.hpp SessionRecording.cpp Scene3DView.hpp Scene3DView.cpp CameraPanel.hpp CameraPanel.cpp EnvironmentDiscovery.hpp EnvironmentDiscovery.cpp)
target_link_libraries(AeroCore_UI PUBLIC AeroCore_Simulation Qt6::Gui Qt6::Widgets)
# Video requires the official Qt Multimedia Widgets addon; all simulation and
# perspective-scene features remain available when it is not installed.
find_package(Qt6 6.4 QUIET COMPONENTS Multimedia MultimediaWidgets)
if(TARGET Qt6::Multimedia AND TARGET Qt6::MultimediaWidgets)
    target_compile_definitions(AeroCore_UI PUBLIC AEROCORE_ENABLE_MULTIMEDIA=1)
    target_link_libraries(AeroCore_UI PUBLIC Qt6::Multimedia Qt6::MultimediaWidgets)
endif()
qt_add_executable(AeroCore WIN32 MACOSX_BUNDLE main.cpp resources.qrc)
target_link_libraries(AeroCore PRIVATE AeroCore_UI)
if(BUILD_TESTING)
    add_executable(lab_types_test tests/lab_types_test.cpp)
    target_link_libraries(lab_types_test PRIVATE AeroCore_Core)
    add_test(NAME lab_types_test COMMAND lab_types_test)
    add_executable(experiment_runner_test tests/experiment_runner_test.cpp)
    target_link_libraries(experiment_runner_test PRIVATE AeroCore_Simulation)
    add_test(NAME experiment_runner_test COMMAND experiment_runner_test)
    add_executable(fleet_transport_test tests/fleet_transport_test.cpp)
    target_link_libraries(fleet_transport_test PRIVATE AeroCore_Simulation)
    add_test(NAME fleet_transport_test COMMAND fleet_transport_test)
    add_executable(fleet_comparison_test tests/fleet_comparison_test.cpp)
    target_link_libraries(fleet_comparison_test PRIVATE AeroCore_UI)
    add_test(NAME fleet_comparison_test COMMAND fleet_comparison_test)
    set_tests_properties(fleet_comparison_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(recording_test tests/recording_test.cpp)
    target_link_libraries(recording_test PRIVATE AeroCore_UI)
    add_test(NAME recording_test COMMAND recording_test)
    set_tests_properties(recording_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(ui_smoke_test tests/ui_smoke_test.cpp resources.qrc)
    target_link_libraries(ui_smoke_test PRIVATE AeroCore_UI)
    add_test(NAME ui_smoke_test COMMAND ui_smoke_test)
    set_tests_properties(ui_smoke_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(scene_3d_test tests/scene_3d_test.cpp)
    target_link_libraries(scene_3d_test PRIVATE AeroCore_UI)
    add_test(NAME scene_3d_test COMMAND scene_3d_test)
    set_tests_properties(scene_3d_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(environment_discovery_test tests/environment_discovery_test.cpp)
    target_link_libraries(environment_discovery_test PRIVATE AeroCore_UI)
    add_test(NAME environment_discovery_test COMMAND environment_discovery_test)
    if(Python3_FOUND)
        add_test(NAME px4_path_discovery_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/px4_path_discovery_test.py)
    endif()
endif()
install(TARGETS AeroCore BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
